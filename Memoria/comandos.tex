\newenvironment{Revision}{%    % Define un nuevo entorno llamado "Revision"
  \ifRevision                  % Si la variable \ifRevision es verdadera...
    %\leavevmode                %   Fuerza a iniciar un párrafo/modo horizontal
    %\begingroup                %   Abre un grupo (para limitar el alcance del color)
    \color{RawSienna}%         %   Cambia el color actual a Turquesa
    %\ignorespaces              %   Ignora los espacios en blanco que sigan
  \else                        % Si la variable \ifRevision es falsa...
    %\leavevmode                %   Fuerza a iniciar un párrafo/modo horizontal
    %\begingroup                %   Abre un grupo (aunque no se use color)
    %\ignorespaces              %   Ignora los espacios en blanco que sigan
  \fi                          % Fin de la condición
}{%                            % Parte de cierre del entorno
    %\endgroup                  %   Cierra el grupo abierto en la apertura
}
