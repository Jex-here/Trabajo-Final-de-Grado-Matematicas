\begin{frame}
    \begin{block}{\centering \large Cifrado de clave simétrica}
        
        \begin{enumerate}
            \item Alice y Bob intercambian una clave \textcolor{blue}{k} mediante un \textcolor{blue}{canal seguro}.
            \item Alice cifra el mensaje \alert{m}, obteniendo \alert{c}. \\
            \item Alice envía el mensaje cifrado \alert{c} por el canal inseguro. \\
            \item Bob recibe \alert{c} y los descifra para obtener \alert{m}.
        \end{enumerate}
        
    \end{block}

\subsection{Cifrado simétrico por bloques}
%橘シェリー橘シェリー橘シェリー橘シェリー橘シェリー橘シェリー橘シェリー


\begin{frame}
   \begin{block}{\centering \large Cifrado por bloques}
        \begin{enumerate}
            \item El mensaje se divide en bloques. \\
            \item Se realizan las mismas operaciones en cada bloque. \\
            \item Se obtiene el texto cifrado de los bloques.
        \end{enumerate}
    \end{block}

    \vspace{1em}

    \centering
    \begin{tikzpicture}[
            >=Stealth,                    % punta de flecha estilizada
            node distance=2cm,
            every node/.style={
                rectangle,
                draw,
                minimum width=2.2cm,
                minimum height=0.8cm,
                align=center,
                font=\small
            }
        ]

        % --- Nodos ---
        % Columna 1: Mensaje (centrado verticalmente)
        \node (msg) at (0,0) {Mensaje};

        % Columna 2: Bloques de texto claro (tres, apilados)
        \node (p1) at (3, 1.8) {Bloque 1};
        \node (p2) at (3, 0)   {Bloque 2};
        \node (p3) at (3,-1.8) {Bloque 3};

        % Columna 3: Bloques de texto cifrado (tres, alineados con los anteriores)
        \node (c1) at (6, 1.8) {Bloque 1 cifrado};
        \node (c2) at (6, 0)   {Bloque 2 cifrado};
        \node (c3) at (6,-1.8) {Bloque 3 cifrado};

        % Columna 4: Texto cifrado final (centrado verticalmente)
        \node (final) at (9,0) {Mensaje cifrado};

        % --- Puntos suspensivos hacia abajo (indicando más bloques) ---
        \node[draw=none] (dots_p) at (3, -2.8) {\vdots};
        \node[draw=none] (dots_c) at (6, -2.8) {\vdots};

        % --- Flechas ---
        % Del mensaje a cada texto claro (apuntando exactamente al borde izquierdo)
        \draw[->] (msg) -- (p1.west);
        \draw[->] (msg) -- (p2.west);
        \draw[->] (msg) -- (p3.west);

        % De cada texto claro a su correspondiente texto cifrado (horizontales)
        \draw[->] (p1) -- (c1);
        \draw[->] (p2) -- (c2);
        \draw[->] (p3) -- (c3);

        % De cada texto cifrado al final (flechas convergentes)
        \draw[->] (c1) -- (final);
        \draw[->] (c2) -- (final);
        \draw[->] (c3) -- (final);

    \end{tikzpicture}
\end{frame}
