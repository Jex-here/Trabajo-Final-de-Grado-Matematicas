\begin{frame}
   \begin{block}{\centering \large Cifrado por bloques}
        \begin{enumerate}
            \item El mensaje se divide en bloques. \\
            \item Se realizan las mismas operaciones en cada bloque. \\
            \item Se obtiene el texto cifrado de los bloques.
        \end{enumerate}
    \end{block}
