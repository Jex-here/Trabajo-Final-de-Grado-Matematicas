\subsection{Qué es un cuerpo finito}
% 蝶々蝶々蝶々蝶々蝶々蝶々蝶々蝶々蝶々蝶々蝶々蝶々蝶々

%=========================ESTRUCTURA DE GRUPO=========================
\begin{frame}%{Estructuras algebraicas}
    \begin{block}{\centering \large Grupo: $(G,\star)$}
        \begin{enumerate}[label=\roman*)]
            \item \textit{Asociatividad}: \hspace{9em}
                $(a \star b) \star c = a \star (b \star c)$
            \item \textit{Existencia de elemento neutro}: \hspace{1.7em}
                $\exists e \hspace{1.12em} \mid  a \star e = a = e \star a$
            \item \textit{Existencia de elementos inversos}: \hspace{0.6em}
                $\forall a \exists b \mid \hspace{0.1em} a \star b  = e = b \star a$
        \end{enumerate}
        Si además se cumple la propiedad de
        \begin{enumerate}[label=iv)]
            \item \textit{Conmutatividad}: \hspace{8.4em}
            $a \star b  = b \star a$
        \end{enumerate}
        se dice que el grupo es un \alert{grupo conmutativo}.
    \end{block} 

    \begin{plainexample}
        \begin{enumerate}[label=--]
            \item $(\N,+)$ no es un grupo.
            \item $(\Z,+)$ es un grupo conmutativo.
            \item $(\R^*,\cdot)$ es un grupo conmutativo.
        \end{enumerate}
    \end{plainexample}
\end{frame}

%=========================ESTRUCTURA DE ANILLO=========================
\begin{frame}%{Estructuras algebraicas}
    \vspace{-0.5em}
    \begin{block}{\centering \large Anillo: $(A,+,\cdot)$}
        \begin{enumerate}[label=\roman*)]
            \item $(A, +)$ es un \textit{grupo conmutativo}.
            \item \textit{Asociatividad para el producto}: \hspace{0.5em}
                $ (a \cdot b) \cdot c = a \cdot (b \cdot c)$
            \item \textit{Propiedad distributiva}:\hspace{4.45em}
                $(a + b) \cdot c = (a \cdot c) + (b \cdot c)$ \\[0.6ex]
                \hspace{14.45em}
                $ c \cdot (a + b) = (c \cdot a) + (c \cdot b)$
        \end{enumerate}
        Si además se cumple que
        \begin{enumerate}[label=\roman*), start=4]
            \item El producto tiene elemento neutro, $A$ es un \alert{anillo unitario}.
            \item El producto es conmutativo, $A$ es un \alert{anillo conmutativo}.
        \end{enumerate}
            Si se cumplen ambas, $A$ es un \alert{anillo unitario conmutativo}.
    \end{block} 
    \vspace{-0.5em}

    \only<1>{
        \begin{plainexample}
            \begin{enumerate}[label=--]
                \item $\Z$ y $\Q$ son anillos unitarios conmutativos.
                \item $\mathcal{M}_n(\R)$ son anillos unitarios.
            \end{enumerate}
        \end{plainexample}
    }
%=========================ANILLO DE POLINOMIOS=========================
    \only<2>{
        \begin{exampleblock}{\centering \large Anillo de polinomios}
            Sea $A$ un anillo unitario conmutativo.\\
            Se definen los polinomios sobre $A$ como el conjunto
            \vspace{-0.25em}
            \[
                A[X] = \{a_0 + a_1X +  \ldots + a_nX^n , \text{ con }  a_i \in A  \text{ y }  n \in \N\}
            \]
            $A[X]$ es un anillo unitario conmutativo.
        \end{exampleblock}
    }
\end{frame}


%=========================UNIDADES=========================
\begin{frame}%{Estructuras algebraicas}
\vspace{5em}
    \begin{block}{\centering \large Unidades}
        Sea $A$ un anillo unitario. Las \alert{unidades} de $A$ son los elementos que tienen inversa respecto del producto.
    \end{block}

    \begin{plainexample}
        \begin{enumerate}[label=--]
            \item Las unidades de $\Z$ son el $1$ y el $-1$.
            \item Las unidades de $\Q$ son todos los elementos de $\Q$ menos el $0$.
            \item Las unidades de $\mathcal{M}_n(\R)$ son las matrices con determinante no nulo.
        \end{enumerate}
    \end{plainexample}
\end{frame}

%=========================CUERPO=========================
\begin{frame}%{Estructuras algebraicas}
    \begin{block}{\centering \large Cuerpo}
        \centering
        Un cuerpo $K$ es un anillo unitario conmutativo verificando que \\
        todo elemento no nulo es unidad.
    \end{block}

    \begin{plainexample}
        \begin{enumerate}[label=--]
            \item $(\Z,+, \cdot)$ no es un cuerpo.
            \item $\Q$, $\R$ y $\C$ son cuerpos con  su suma y producto usual.
            \item Los anillos de polinomios no son cuerpos.
        \end{enumerate}
    \end{plainexample}
\end{frame}

\subsection{Cómo obtener cuerpos finitos}

%=========================IDEAL=========================
\begin{frame}
    \begin{block}{\centering \large Ideal}
        Sea $A$ un anillo unitario conmutativo.\\
        Un ideal $I$ de $A$ es un subconjunto de $A$ que satisface:
    \begin{enumerate}[label=\roman*)]
        \item $I$ es subgrupo de (A,+).
        \item Los elementos de $I$ son absorbentes en $A$ para el producto, es decir, para todo $i \in I, a\in A$ se cumple $i\cdot a \in I$
    \end{enumerate}
    \end{block}

    \begin{plainexample}
        \begin{enumerate}[label=--]
            \item Los ideales de $\Z$ son de la forma $n\Z:= \{ n \cdot x\mid x\in\Z\}$.\\[1ex]
                Se dice entonces que el ideal está generado por $n$, y se denota como
                \[(n) = n\Z\]
        \end{enumerate}
    \end{plainexample}
\end{frame}

%=========================ANILLO COCIENTE=========================
\begin{frame}
    \begin{block}{\centering \large Anillos cocientes}
        Sea $A$ un anillo unitario conmutativo e $I$ un ideal de $A$.\\
        Se define en $A$ la relación de equivalencia 
            \vspace{-0.5em}
            \[x \sim y \iff x -y \in I.\]
        El conjunto cociente $A/I$ es un anillo unitario conmutativo con la suma y producto definidos como
        \[[x] + [y] = [x + y]\]
        \[[x] \cdot [y] = [x \cdot y]\]
    \end{block}

    \begin{plainexample}
        \begin{enumerate}[label=--]
            \setlength{\itemsep}{-0.2em}
            \item El ideal $(n)$ produce el cociente $\Z/(n)$:
                \vspace{-0.5em}
                \[\Z_n = \{0,1, \ldotp, n-1\}\]
            % \item En el anillo cociente $\Z_6$ se verifica la identidad
            %     \vspace{-0.5em}
            %     \[2 \cdot 3 = 0\]
        \end{enumerate}
    \end{plainexample}
\end{frame}

%=========================IDEAL MAXIMAL=========================
\begin{frame}
    \begin{block}{\centering \large Ideal maximal}
        Sea $A$ un anillo unitario conmutativo e $I$ un ideal de $A$.\\[0.3ex]
        $I$ es un ideal maximal si no existe otro ideal distinto a $A$ que lo contenga estrictamente.
    \end{block}

    \begin{plainexample}
        \begin{enumerate}[label=--]
            \item En $\Z$, el ideal generado por 6: (6), no es maximal, pues está contenido en los ideales $(2)$ y $(3)$.
            \item En $\Z$, los ideales maximales están generados por números primos.
        \end{enumerate}
    \end{plainexample}

    \begin{prop}
        \centering $I$ es maximal $\iff A/I$ es un cuerpo
    \end{prop}

    \begin{plainexample}
        \begin{enumerate}[label=--]
            \item Tomando los números primos en $\Z$, se obtienen los cuerpos finitos 
            \vspace*{-1.7em}
            \begin{align*}
                \Z_2 &= \{0,1\}\\
                \Z_3 &= \{0,1,2\}\\
                \Z_p &= \{0,1,2, \ldots, p-1\}
            \end{align*}
        \end{enumerate}
    \end{plainexample}
\end{frame}

%=========================DOMINIO EUCLÍDEO=========================
\begin{frame}
    \begin{block}{\centering \large Dominio euclídeo}
        Un dominio euclídeo es un anillo unitario conmutativo, $A$, sin divisores de 0, y donde existe una función norma $\norm{\cdot} \colon A^* \to \N$, verificando:
        \begin{enumerate}[label=\roman*)]
            \item Para todo $x,y\in A^*$ \[\norm{x} \leq\norm{xy}\]
            \item Para todo $x,y \in A^*$, existen $q,r \in A$, tales que 
                \[y = qx + r,\]
                donde $r = 0 \text{ o bien } \norm{r} < \norm{x}$.
        \end{enumerate}
    \end{block}

    \begin{prop}
        \centering Si $K$ es un cuerpo $\Longrightarrow K[X]$ es un dominio euclídeo 
    \end{prop}

    \begin{plainexample}
        \begin{enumerate}[label=--]
            \item En $\Z$, el valor absoluto $|x|$ es una norma.
            \item En $K[X]$, el grado de los polinomios es una norma.
        \end{enumerate}
    \end{plainexample}
\end{frame}

%=========================CUERPOS FINITOS=========================
\begin{frame}
    \begin{prop}
        Un polinomio $f$ de grado $n$ que genera un ideal maximal en $\Z_p[X]$ produce el cuerpo:
        \[ \Z_p[X]/(f) = \{a_0 + a_1X +  \ldots + a_nX^{n-1} , \  a_i \in \Z_p\},\]
        formado por $p^n$ elementos.
    \end{prop}

    \begin{plainexample}
        \begin{enumerate}[label=--]
            \item Tomando $p = 2$ y $f = X^2+X+1$ se obtiene el cuerpo
            \[\Z_2[X]/(f) = \{0,\ 1,\ X,\ X+1\}\]
            \item Tomando $p = 3$ y $g = X^2 + X + 2$ se obtiene el cuerpo
            \begin{align*}
                \Z_3[X]/(g) \quad = &&\{  &0   ,&& 1   ,&& 2  ,   &&&&\\
                &&    &   X,&& X+1 ,&& X+2,   &&&&\\
                &&    &  2X,&& 2X+1,&& 2X+2\} &&&&
            \end{align*}
        \end{enumerate}
    \end{plainexample}
\end{frame}

%=========================CARACTERIZACIÓN CUERPOS FINITOS=========================
\begin{frame}
    \vspace{8em}
    \begin{teo}{Caracterización de cuerpos finitos}
        \vspace{0.5em}
        \begin{enumerate}[label=--]
            \item Solo existen cuerpos finitos con $p^n$ elementos.
            \item Existe un único cuerpo con $p^n$ elementos.
        \end{enumerate}
    \end{teo}
\end{frame}

%=========================BÉZOUT=========================
\begin{frame}
    \begin{block}{\centering \large Búsqueda de inverso}
        Dado $x\in \Z_p[X]/(f)$, se ha de buscar $y\in \Z_p[X]/(f)$ tal que
        \setlength{\belowdisplayskip}{-1pt}   % Reduce espacio abajo
        \vspace{-0.5em}
        \begin{align*}
            \setlength{\itemsep}{-0.3em}
            xy      &= 1    \\
            [x][y]  &= [1]  \\
            xy      &\sim 1       \\
            xy - 1  &= fg'    \\
            x\alert{y} + f\alert{g}  &= 1    
        \end{align*}

    \end{block}

    \begin{prop}
        La identidad de Bézout \[x\alert{y} + f\alert{g}  = 1\] es resoluble en un dominio euclídeo cuando  $\mcd{x,f} = 1$.
    \end{prop}

\end{frame}
